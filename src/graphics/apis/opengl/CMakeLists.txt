cmake_minimum_required(VERSION 3.10 FATAL_ERROR)

add_library(engine_opengl_backend)

target_sources(
  engine_opengl_backend
  PUBLIC FILE_SET
         CXX_MODULES
         BASE_DIRS
         ${PROJECT_SOURCE_DIR}
         FILES
         engine.graphics.apis.opengl.cppm
         engine.graphics.apis.opengl.buffers.cppm
         engine.graphics.apis.opengl.shader_utils.cppm
         engine.graphics.apis.opengl.groups.cppm
         engine.graphics.apis.opengl.shader_gen.cppm)

FetchContent_Declare(
  glfw
  GIT_REPOSITORY https://github.com/glfw/glfw
  GIT_TAG 3.5.1)

FetchContent_MakeAvailable(glfw)

FetchContent_Declare(
  EnTT
  GIT_REPOSITORY https://github.com/skypjack/entt
  GIT_TAG v3.16.0)

FetchContent_MakeAvailable(EnTT)

FetchContent_Declare(
  glm
  GIT_REPOSITORY https://github.com/g-truck/glm
  GIT_TAG 1.0.3)

FetchContent_MakeAvailable(glm)

# glad needs to be generated by the user

target_link_libraries(
  engine_opengl_backend
  PRIVATE glfw EnTT::EnTT engine_ecs
  PUBLIC glm::glm)
target_include_directories(engine_opengl_backend
                           PRIVATE ${PROJECT_SOURCE_DIR}/include)
target_compile_features(engine_opengl_backend PRIVATE cxx_std_20)
