const BasicMeshIndexed create_basic_circle(GLuint &vao, GLuint &vbo,
                                           GLuint &ebo, const GLint pos_index,
                                           const GLint vbo_binding_index,
                                           const int vertex_count) {
  std::vector<GLfloat> vertices;
  vertices.reserve(vertex_count + 1);

  std::vector<GLuint> indices;
  indices.reserve(vertex_count * 3);

  double theta = 0.0;
  const double inc = (2.0 * PI) / static_cast<double>(vertex_count);

  // center - index 0
  vertices.push_back(0.0f);
  vertices.push_back(0.0f);

  float k = 1;

  while (theta < 2.0 * PI) {
    // x
    vertices.push_back(static_cast<float>(std::cos(theta)));
    // y
    vertices.push_back(static_cast<float>(std::sin(theta)));

    if (theta == 2.0 * PI - inc) {
      indices.push_back(k);
      indices.push_back(0);
      indices.push_back(1);
    } else {
      indices.push_back(k);
      indices.push_back(0);
      indices.push_back(k + 1);
    }

    theta += inc;
    ++k;
  }

  glCreateBuffers(1, &vbo);
  glNamedBufferStorage(vbo, vertices.size() * sizeof(GLfloat), vertices.data(),
                       0);

  glCreateBuffers(1, &ebo);
  glNamedBufferStorage(ebo, indices.size() * sizeof(GLuint), indices.data(), 0);

  set_attrib(vao, pos_index, vbo_binding_index, 2, GL_FLOAT, GL_FALSE, 0, 0);

  glVertexArrayVertexBuffer(vao, vbo_binding_index, vbo, 0,
                            2 * sizeof(GLfloat));
  glVertexArrayElementBuffer(vao, ebo);

  return BasicMeshIndexed{indices.size(), vao};
};
